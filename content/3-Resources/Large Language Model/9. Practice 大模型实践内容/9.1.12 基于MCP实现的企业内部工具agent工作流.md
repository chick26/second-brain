> 首先来解析`mcp_server.py` ，其采用 Python 标准库中的 `http.server`，通过继承 `BaseHTTPRequestHandler` 实现对 POST 请求的统一处理。核心在于解析 HTTP 请求体中的 JSON-RPC 2.0 结构，分发到不同方法并返回 JSON。
> 下面展示了如何读取请求、解析 JSON-RPC 字段、构造响应模板并调用 `_send_response`。这部分框架在大多数基于 JSON-RPC 的 HTTP 服务中都会存在：读取 `method` 字段决定走向，若未命中则返回 “`Method not found`”。

```python
from http.server import BaseHTTPRequestHandler, HTTPServer
import json

class MCPRequestHandler(BaseHTTPRequestHandler):
    def _send_response(self, content):
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        # 将 Python 对象转为 JSON 字符串写回
        self.wfile.write(json.dumps(content).encode('utf-8'))

    def do_POST(self):
        # 读取请求体长度并解析 JSON
        content_length = int(self.headers['Content-Length'])
        post_data = self.rfile.read(content_length)
        request_body = json.loads(post_data)

        method = request_body.get("method")
        params = request_body.get("params", {})
        request_id = request_body.get("id")

        print(f"\n[Server] Received method: {method} with params: {params}")

        # 构造基本响应结构
        response = {"jsonrpc": "2.0", "id": request_id}

        # 根据 method 进行分发
        if method == "mcp/discover":
            # 返回能力清单
            response["result"] = { ... }
        elif method == "mcp/run":
            # 执行具体能力
            ...
        else:
            response["error"] = {"code": -32601, "message": "Method not found"}

        self._send_response(response)

```

## LLM 驱动工作流 Agent

下面解析 `mcp_agent_llm.py` 的关键部分，并在必要处展示代码片段。

1. **通用占位符解析函数**

Agent 需要将 LLM 生成的包含占位符的参数格式化为实际值，`resolve_placeholders` 实现了简单的占位符替换逻辑，支持从初始请求数据或此前步骤结果中读取字段。示例代码如下：

```python
import re

def resolve_placeholders(value: str, initial_request_data: dict, workflow_context: dict):"""
    支持形式：
      {{request.key}}, {request.key}
      {{steps[1].output.asset_id}}, {steps.1.asset_id} 等
    """
    if not isinstance(value, str):
        return value
    raw = value.strip()# 处理 {{...}} 或 {...}
    if raw.startswith("{{") and raw.endswith("}}"):
        placeholder = raw[2:-2].strip()
    elif raw.startswith("{") and raw.endswith("}"):
        placeholder = raw[1:-1].strip()
    else:
        return value  # 非占位符，直接返回原值

    # request.xxx
    if placeholder.startswith("request."):
        _, req_key = placeholder.split('.', 1)
    return initial_request_data.get(req_key)
# steps[...] 或 steps.N
    if placeholder.startswith("steps"):# 支持 steps[N].field 或 steps.N.field
        m = re.match(r"^steps\[(\d+)\](?:.(.*))?$", placeholder)
        if m:
            idx = int(m.group(1))
            rest = m.group(2) or ""
        else:
            m2 = re.match(r"^steps\.(\d+)(?:\.(.*))?$", placeholder)
            if m2:
                idx = int(m2.group(1))
                rest = m2.group(2) or ""
            else:
                return None
        steps_list = workflow_context.get("steps", [])
        if idx < 0 or idx >= len(steps_list):
            return None
        prev_output = steps_list[idx].get("output", {})
        if not rest:
            return prev_output
        if rest.startswith("output."):
            rest = rest[len("output."):]# 链式访问
        resolved = prev_outputfor subkey in rest.split('.'):
        if isinstance(resolved, dict):
                resolved = resolved.get(subkey)
        else:
            resolved = getattr(resolved, subkey, None)
            if resolved is None:
                break
        return resolved
return None
```

* **示例说明**：当参数值为 `"{{request.employee_id}}"` 时，会从 `initial_request_data["employee_id"]` 中获取实际值。若为 `"{steps[0].output.asset_id}"`，则在执行到第 2 步后，从 `workflow_context["steps"][0]["output"]["asset_id"]` 拿到先前结果。若占位符无法解析（格式不符合或索引越界），函数返回 `None` 并在日志中可打印警告，Agent 可据此决定是否终止或继续。
- **MCP 客户端封装**

封装 HTTP 请求与 JSON-RPC 交互细节，示例代码如下：

```python
import requests
import json
import uuid

class MCPClient:def init(self, server_url):
        self.server_url = server_url
        self.headers = {'Content-Type': 'application/json'}
def _send_request(self, method, params=None):
        payload = {"jsonrpc": "2.0", "method": method, "params": params or {}, "id": str(uuid.uuid4())}print(f"\n[Agent] --> Sending request to server: {json.dumps(payload, indent=2)}")try:
            response = requests.post(self.server_url, data=json.dumps(payload), headers=self.headers)
            response.raise_for_status()
            response_json = response.json()if "error" in response_json:raise Exception(f"Server Error: {response_json['error']['message']}")print(f"[Agent] <-- Received result: {json.dumps(response_json.get('result'), indent=2)}")return response_json.get("result")except Exception as e:print(f"[Agent] An error occurred: {e}")return None

    def discover(self):return self._send_request("mcp/discover")
def run(self, capability, parameters):
        params = {"capability": capability, "parameters": parameters}return self._send_request("mcp/run", params)
```

* **Prompt 设计与 JSON 输出**

关键方法 `_generate_plan_with_llm` 将自然语言请求与可用能力传入 LLM，要求输出 JSON 数组，每个对象包含 `capability` 与 `parameters` 字段。示例代码：

```python
def _generate_plan_with_llm(self, user_request):print("\n[Agent-Brain] Asking LLM to generate a plan...")
    tools_for_prompt = json.dumps(self.capabilities, indent=2)
    system_prompt = f"""
You are a highly intelligent workflow orchestration agent. Your task is to create a step-by-step execution plan based on a user's request and a list of available tools.
You must respond with a valid JSON array of objects, where each object represents a step in the plan.
Each step object must have two keys:
1. "capability": The exact name of the tool to use from the provided list.
2. "parameters": An object containing the parameters needed for that tool.
The parameters can be:
- A literal value (e.g., a string or number).
- A placeholder string that references data from the initial user request, formatted as `{{request.key}}`.
- A placeholder string that references the output of a previous step, formatted as `{{steps[N].output.key}}`, where N is the zero-based index of the step.
Here is the list of available tools:
{tools_for_prompt}
Now, create a plan for the following user request.
"""
    response = self.llm_client.chat.completions.create(
        model="qwen-plus",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_request}
        ],
        response_format={"type": "json_object"}
    )
    plan_json = response.choices[0].message.contentprint(f"[Agent-Brain] LLM generated the following plan:\n{plan_json}")return json.loads(plan_json)
```

* **要点**：
  * 将能力列表以 JSON 形式嵌入 Prompt，方便 LLM 理解工具集合及参数规范。
  * 强制 `response_format={"type": "json_object"}` 令 LLM 返回可解析的 JSON。
  * 生产环境中需补充对 LLM 返回的 JSON 验证：检查是否为数组、各 step 的 `capability` 是否在已知列表中、`parameters` 是否为对象、字段匹配与必需字段检查等，防止后续执行出错。
- **上下文管理与参数替换**

执行阶段由 `_execute_plan` 实现，将每一步结果保存到 `workflow_context` 以供后续占位符解析。示例代码：

```python
def _execute_plan(self, plan, initial_request_data):
    self.workflow_context = {"request": initial_request_data, "steps": []}for i, step in enumerate(plan):print(f"[Agent-Executor] Executing Step {i+1}: {step['capability']}")
        resolved_params = {}for key, value in step["parameters"].items():print(f"[DEBUG] 参数 key={key}, 原始 value={repr(value)}")
            resolved = resolve_placeholders(value, initial_request_data, self.workflow_context)print(f"[DEBUG] 解析后 resolved={repr(resolved)}")if isinstance(value, str) and (value.strip().startswith("{") and value.strip().endswith("}")):if resolved is None:print(f"[Agent-Executor] WARNING: 占位符 '{value}' 解析失败或结果 None")
                resolved_params[key] = resolvedelse:
                resolved_params[key] = valueprint(f"[Agent-Executor] Resolved parameters: {resolved_params}")try:
            result = self.mcp_client.run(step["capability"], resolved_params)except Exception as e:print(f"[Agent-Executor] 调用 MCP 出错: {e}")return False
        if not result:print(f"[Agent-Executor] FATAL: Step {i+1} failed. Aborting workflow.")return False
            self.workflow_context["steps"].append({"output": result})print(f"[Agent-Executor] Step {i+1} completed successfully. workflow_context steps count={len(self.workflow_context['steps'])}")
     return True
```

* **核心流程**：
  1. 初始化 `workflow_context = {"request": initial_request_data, "steps": []}`。
  2. 遍历 plan，每步打印日志、依次对 `parameters` 进行占位符解析：若值符合 `{...}` 或 `{{...}}` 格式，则调用 `resolve_placeholders`；否则按字面值传递。
  3. 调用 `self.mcp_client.run` 执行能力；若抛异常或返回 `None`/失败值，则中止。
  4. 将每步结果以 `{"output": result}` 形式追加到 `workflow_context["steps"]`。
* **改进建议**：对解析失败的占位符，可选择主动抛错并终止流程，也可提供默认值或回退逻辑；对返回结果应做 schema 校验或内容完整性检查；对于长流程，可引入超时控制、并行或分支管理。

1. 严格的 Plan 验证

在 `_generate_plan_with_llm` 获取 JSON 后，应做验证。例如：

import jsonschema

step_schema = {"type": "object","properties": {"capability": {"type": "string"},"parameters": {"type": "object"}
    },"required": ["capability","parameters"]
}

# 在解析后进行校验
try:

def run_with_retry(self, capability, params, retries=3):
    for attempt in range(retries):
        result = self.mcp_client.run(capability, params)
        if result is not None:
            return result
        time.sleep(2 ** attempt)
        raise RuntimeError(f"Capability {capability} failed after {retries} retries")

* **安全与权限控制**

示例代码未做鉴权，生产场景须对每次请求进行认证和鉴权，例如：

* Agent 身份校验：携带 API Token 或 OAuth 方式；
* 对某些能力进行权限限制：只有特定角色或服务才能调用；
* 在能力执行处增加审计日志，记录谁在何时以何参数调用了哪些操作，支持事后审计与合规。

在 `MCPRequestHandler` 中，可在 `do_POST` 开头加入身份校验逻辑：

```python
auth_header = self.headers.get("Authorization")
if not valid_token(auth_header):
    self.send_response(401)
    self.end_headers()return
```

* **LLM Prompt 优化**

当前 Agent 直接将自然语言请求与能力列表传给 LLM，若生成计划失败或不合理，可考虑：

* 给 LLM 追加上下文示例（few-shot 示例）：展示常见请求如何映射到步骤序列，帮助 LLM 理解结构。
* 在生成计划后，若校验失败，可将错误信息反馈给 LLM，请其修正。例如：

```python
if validation_error:
    correction_prompt = f"上次生成的计划有问题：{validation_error}。请基于原始请求重新生成正确的计划。"
    # 再次调用 LLM
```

* 对长对话或复杂需求，可让 Agent 与用户进行交互确认关键决策点，如询问选择哪种资产类型、审批人等，再据用户回答修改 plan。
- **多后端集成与能力编排**

当需要集成更多系统（如财务系统、审批流程系统、CRM 等），可在 MCP Server 端或中间层统一桥接各系统接口，能力描述中注明所需权限与依赖关系。Agent 在生成 plan 时，可读取能力的“前置需求”信息，避免顺序错误或遗漏。例如，在能力 Schema 中增加字段：

```json
"iam_create_user_accounts": {
  "description": "...",
  "parameters": { ... },
  "returns": { ... },
  "prerequisites": ["hris_create_employee_profile"]
}
```

在生成或执行前校验，确保先执行员工档案创建再创建账户。

* **占位符解析示例改进**

若某一步需要将前一步输出字段改名或组合，可在 plan 中直接指定复杂字符串，然后 Agent 解析。例如：

```json
{
  "capability": "notifier_send_notification",
  "parameters": {
    "recipient": "{steps[3].output.email}",
    "message": "您好，{request.full_name}，您的入职流程已完成，邮箱：{steps[3].output.email}，Jira 账号：{steps[3].output.jira_account}"
  }
}
```

`resolve_placeholders` 会按需要多次调用，将 `{steps[3].output.email}`、`{request.full_name}` 等替换为实际值。若需要更复杂模板（如日期格式化），可在解析前使用 Python 的 `str.format` 或自定义模板引擎；但需注意安全和异常处理。
